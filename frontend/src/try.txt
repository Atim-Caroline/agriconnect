import sys
sys.setrecursionlimit(1 << 25)

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    p = [0] * (n + 1)
    a = [0] * (n + 1)
    b = [0] * (n + 1)
    children = [[] for _ in range(n + 1)]
    for i in range(1, n + 1):
        p[i] = int(next(it))
        a[i] = int(next(it))
        b[i] = int(next(it))
        if p[i] != 0:
            children[p[i]].append(i)

    def dfs(u):
        # returns list of achievable (sum_a, sum_b) for valid sets in the subtree of u that include u
        res = [(a[u], b[u])]
        for v in children[u]:
            sub = dfs(v)
            new_res = []
            for xa, xb in res:
                # do not take anything from this child
                new_res.append((xa, xb))
                # take one of the configurations from the child
                for sa, sb in sub:
                    new_res.append((xa + sa, xb + sb))
            res = new_res
        return res

    candidates = dfs(1)
    ans = max(x * y for x, y in candidates)
    print(ans)

if __name__ == "__main__":
